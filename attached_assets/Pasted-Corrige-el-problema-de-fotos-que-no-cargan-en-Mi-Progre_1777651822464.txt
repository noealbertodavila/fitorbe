function mostrarFoto(url, contenedor, label) {
  if (!url) {
    contenedor.innerHTML = `

        <span style="font-size:24px;">📷</span>
        <span style="font-size:11px;">${label}</span>
      </div>`;
    return;
  }
  
  const img = document.createElement('img');
  img.src = url;
  img.style.cssText = 'width:100%;aspect-ratio:3/4;object-fit:cover;border-radius:8px;';
  img.alt = label;
  img.onerror = function() {
    this.parentElement.innerHTML = `
      <div style="width:100%;aspect-ratio:3/4;background:#0F1A14;border:0.5px dashed #EF4444;border-radius:8px;display:flex;align-items:center;justify-content:center;color:#EF4444;flex-direction:column;gap:8px;">
        <span style="font-size:24px;">⚠️</span>
        <span style="font-size:11px;">Foto no disponible</span>
      </div>`;
  };
  contenedor.innerHTML = '';
  contenedor.appendChild(img);
}
