      .replace(/```json/g, '')
      .replace(/```/g, '')
      .trim();
    
    // Intentar parsear
    let planNutricion;
    try {
      planNutricion = JSON.parse(jsonLimpio);
    } catch(parseError) {
      console.error('Error parseando JSON de nutrición:', parseError.message);
      console.error('Posición del error:', parseError.message.match(/position (\d+)/)?.[1]);
      console.error('Respuesta del AI (primeros 500 chars):', jsonLimpio.substring(0, 500));
      
      return res.status(500).json({ 
        error: 'Error al procesar el plan generado. Intenta de nuevo.',
        detalle: parseError.message
      });
    }
    
    res.json(planNutricion);
    
  } catch(e) {
    console.error('Error generando nutrición:', e);
    res.status(500).json({ error: 'Error al generar el plan: ' + e.message });
  }
}
