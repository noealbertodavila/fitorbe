1. Agregar ruta estática para servir las fotos de progreso:

const express = require('express');

// Servir archivos estáticos de uploads
app.use('/uploads', express.static(path.join(__dirname, 'uploads')));

// Verificar que la carpeta existe, si no crearla:
const fs = require('fs');
const carpetasNecesarias = [
  './uploads',
  './uploads/progreso',
  './uploads/perfil'
];
carpetasNecesarias.forEach(carpeta => {
  if (!fs.existsSync(carpeta)) {
    fs.mkdirSync(carpeta, { recursive: true });
    console.log('Carpeta creada:', carpeta);
  }
});

2. Verificar que multer guarda las fotos en la ruta correcta:
const storage = multer.diskStorage({
  destination: function(req, file, cb) {
    const clienteId = req.body.cliente_id || req.usuario.id;
    const fecha = new Date().toISOString().slice(0, 7); // 2026-05
    const carpeta = path.join(__dirname, 'uploads', 'progreso', String(clienteId), fecha);
    
    // Crear carpeta si no existe
    fs.mkdirSync(carpeta, { recursive: true });
    cb(null, carpeta);
  },
  filename: function(req, file, cb) {
    const nombre = file.fieldname + '_' + Date.now() + path.extname(file.originalname);
    cb(null, nombre);
  }
});
