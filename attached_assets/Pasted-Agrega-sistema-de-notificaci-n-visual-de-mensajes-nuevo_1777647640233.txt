4. PARPADEO DEL FAVICON (ícono de la pestaña):
Alternar entre ícono normal y ícono con punto rojo cada 2 segundos cuando hay mensajes:
let parpadeando = false;
function parpadearFavicon(hayMensajes) {
  if (!hayMensajes) {
    document.querySelector("link[rel='icon']").href = '/favicon-normal.ico';
    return;
  }
  setInterval(() => {
    const favicon = document.querySelector("link[rel='icon']");
    parpadeando = !parpadeando;
    favicon.href = parpadeando ? '/favicon-notif.ico' : '/favicon-normal.ico';
  }, 2000);
}

5. VERIFICACIÓN AUTOMÁTICA CADA 30 SEGUNDOS:
async function verificarMensajesNuevos() {
  const token = localStorage.getItem('nahofitToken');
  const usuario = JSON.parse(localStorage.getItem('nahofitUsuario') || '{}');
  
  try {
    const res = await fetch('/api/mensajes/no-leidos', {
      headers: { Authorization: `Bearer ${token}` }
    });
    const data = await res.json();
    const cantidad = data.cantidad || 0;
    
    // Actualizar badge del menú
    const badge = document.getElementById('badge-mensajes');
    if (badge) {
      badge.textContent = cantidad;
      badge.style.display = cantidad > 0 ? 'flex' : 'none';
    }
    
    // Mostrar/ocultar banner
    const banner = document.getElementById('banner-mensaje');
    if (banner) {
      banner.style.display = cantidad > 0 ? 'flex' : 'none';
      const texto = banner.querySelector('.texto-banner');
      if (texto) {
        texto.textContent = cantidad === 1 
          ? 'Nahomy te envió un mensaje' 
          : `Tienes ${cantidad} mensajes nuevos de Nahomy`;
      }
      const numBadge = banner.querySelector('.num-badge');
      if (numBadge) numBadge.textContent = cantidad;
    }
    
    // Actualizar título pestaña
    document.title = cantidad > 0 
      ? `(${cantidad}) Mensaje nuevo — NAHOFIT.LIFE` 
      : 'NAHOFIT.LIFE';
      
  } catch(e) {
    console.log('Error verificando mensajes:', e);
  }
}

// Verificar al cargar y cada 30 segundos
verificarMensajesNuevos();
setInterval(verificarMensajesNuevos, 30000);
