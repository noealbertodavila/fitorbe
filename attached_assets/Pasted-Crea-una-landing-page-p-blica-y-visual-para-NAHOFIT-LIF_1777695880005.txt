let idiomaActual = localStorage.getItem('nahofit_idioma') || 'es';

function cambiarIdioma(lang) {
  idiomaActual = lang;
  localStorage.setItem('nahofit_idioma', lang);
  document.querySelectorAll('[data-texto]').forEach(el => {
    const key = el.dataset.texto;
    if (textos[lang][key]) el.textContent = textos[lang][key];
  });
  document.querySelectorAll('.lang-btn').forEach(b => {
    b.classList.toggle('activo', b.dataset.lang === lang);
  });
}
