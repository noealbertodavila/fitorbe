function mostrarEstadoVacio(seccion) {
  const mensajes = {
    rutina: { icono: '💪', titulo: 'Sin rutina esta semana', desc: 'Tu coach aún no ha asignado tu plan de entrenamiento.' },
    nutricion: { icono: '🥗', titulo: 'Sin plan de nutrición', desc: 'Tu coach aún no ha asignado tu plan de alimentación.' },

    emocional: { icono: '🧠', titulo: 'Sin check-ins', desc: 'Completa tu primer check-in emocional.' },
    mensajes: { icono: '💬', titulo: 'Sin mensajes', desc: 'Escríbele a tu coach para comenzar.' },
    pagos: { icono: '💳', titulo: 'Sin pagos', desc: 'No hay pagos en tu historial aún.' }
  };
  const m = mensajes[seccion];
  return `
    <div style="text-align:center; padding: 60px 20px; color: var(--texto-secundario);">
      <div style="font-size:48px; margin-bottom:16px;">${m.icono}</div>
      <h3 style="color: var(--texto-principal); margin-bottom:8px;">${m.titulo}</h3>
      <p style="margin-bottom:24px;">${m.desc}</p>
    </div>`;
}
