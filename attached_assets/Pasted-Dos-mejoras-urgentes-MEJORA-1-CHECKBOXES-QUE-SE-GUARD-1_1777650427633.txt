MEJORA 1 — CHECKBOXES QUE SE GUARDAN
═══════════════════════════════════════

Cuando el cliente marca ejercicios como completados, guardar inmediatamente en la base de datos para que persistan al cambiar de sección.

1. Cada checkbox al hacer clic llama a esta función:
async function marcarEjercicio(ejercicioIndex, completado) {
  const token = localStorage.getItem('nahofitToken');
  const usuario = JSON.parse(localStorage.getItem('nahofitUsuario'));
  const hoy = new Date().toISOString().split('T')[0];
  
  // Guardar en localStorage inmediatamente (respuesta visual instantánea)
  const key = `checkin_${usuario.id}_${hoy}`;
  let completados = JSON.parse(localStorage.getItem(key) || '[]');
  if (completado) {
    if (!completados.includes(ejercicioIndex)) completados.push(ejercicioIndex);
  } else {
    completados = completados.filter(i => i !== ejercicioIndex);
  }
  localStorage.setItem(key, JSON.stringify(completados));
  
  // Actualizar barra de progreso inmediatamente
  const totalEjercicios = document.querySelectorAll('.ejercicio-item').length;
  actualizarBarraProgreso(completados.length, totalEjercicios);
  
  // Guardar en base de datos en segundo plano
  try {
    await fetch('/api/checkin/ejercicio', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${token}`
      },
      body: JSON.stringify({
        cliente_id: usuario.id,
        fecha: hoy,
        ejercicio_index: ejercicioIndex,
        completado: completado
      })
    });
  } catch(e) {
    console.log('Error guardando en BD, pero quedó en localStorage:', e);
  }
}

2. Al cargar Mi Rutina, restaurar estado guardado:
async function restaurarEstadoEjercicios() {
  const usuario = JSON.parse(localStorage.getItem('nahofitUsuario'));
  const hoy = new Date().toISOString().split('T')[0];
  const key = `checkin_${usuario.id}_${hoy}`;
  
  // Primero usar localStorage para respuesta inmediata
  let completados = JSON.parse(localStorage.getItem(key) || '[]');
  
  // Luego verificar con el servidor
  try {
    const token = localStorage.getItem('nahofitToken');
    const res = await fetch(`/api/checkin/ejercicios/${usuario.id}/${hoy}`, {
      headers: { Authorization: `Bearer ${token}` }
    });
    if (res.ok) {
      const data = await res.json();
      completados = data.ejercicios_completados || completados;
      localStorage.setItem(key, JSON.stringify(completados));
    }
  } catch(e) {
    console.log('Usando localStorage como fallback');
  }
  
  // Aplicar estado a los checkboxes
  completados.forEach(index => {
    const card = document.querySelector(`[data-ejercicio-index="${index}"]`);
    if (card) {
      const checkbox = card.querySelector('input[type="checkbox"]');
      if (checkbox) checkbox.checked = true;
      card.classList.add('completado');
      card.style.opacity = '0.7';
      card.querySelector('.nombre-ejercicio')?.classList.add('tachado');
    }
  });
  
  // Actualizar barra de progreso
  const total = document.querySelectorAll('.ejercicio-item').length;
  actualizarBarraProgreso(completados.length, total);
}
