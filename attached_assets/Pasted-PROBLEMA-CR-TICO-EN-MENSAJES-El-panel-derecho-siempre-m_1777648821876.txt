3. La función cargarConversacion DEBE limpiar el panel antes de cargar:
async function cargarConversacion(clienteId, clienteNombre) {
  const token = localStorage.getItem('nahofitToken');
  
  // LIMPIAR panel derecho inmediatamente
  const panel = document.getElementById('panel-conversacion');
  panel.innerHTML = `
    <div style="display:flex;align-items:center;justify-content:center;height:200px;">
      <div style="color:#4B7A5E;">Cargando mensajes de ${clienteNombre}...</div>
    </div>`;
  
  // Marcar item activo
  document.querySelectorAll('.conversacion-item').forEach(i => i.classList.remove('activo'));
  document.querySelector(`[data-cliente-id="${clienteId}"]`)?.classList.add('activo');
  
  try {
    const res = await fetch(`/api/mensajes/conversacion/${clienteId}`, {
      headers: { Authorization: `Bearer ${token}` }
    });
    
    if (!res.ok) throw new Error('Error al cargar');
    const mensajes = await res.json();
    
    // Verificar en consola que son los mensajes correctos
    console.log(`Mensajes de ${clienteNombre} (ID:${clienteId}):`, mensajes.length, 'mensajes');
    
    renderizarConversacion(clienteId, clienteNombre, mensajes);
    
  } catch(e) {
    panel.innerHTML = `
      <div style="text-align:center;padding:40px;color:#EF4444;">
        Error al cargar mensajes de ${clienteNombre}
      </div>`;
  }
}

async function enviarMensaje(receptorId) {
  const input = document.getElementById('input-mensaje');
  const contenido = input.value.trim();
  if (!contenido) return;
  
  const token = localStorage.getItem('nahofitToken');
  input.value = '';
  
  try {
    await fetch('/api/mensajes', {
      method: 'POST',
      headers: { 
        'Content-Type': 'application/json',
        Authorization: `Bearer ${token}` 
      },
      body: JSON.stringify({ receptor_id: receptorId, contenido })
    });
    
    // Recargar conversación
    const nombreCliente = document.querySelector(`[data-cliente-id="${receptorId}"]`)?.dataset.clienteNombre || '';
    cargarConversacion(receptorId, nombreCliente);
    
  } catch(e) {
    console.error('Error enviando mensaje:', e);
  }
}
