PROBLEMA 1 — DÍAS "FALLADO" SIN DATOS
═══════════════════════════════════════

En el calendario semanal, días sin rutina asignada muestran "FALLADO" en rojo. Esto es incorrecto.

REGLA DEFINITIVA:
- FALLADO rojo = tenía rutina asignada Y no la completó
- DESCANSO gris = NO tenía rutina asignada ese día
- NUNCA mostrar FALLADO si no había plan

CORRECCIÓN en la lógica del calendario:
async function obtenerEstadoDia(clienteId, fecha) {
  const token = localStorage.getItem('nahofitToken');
  
  // Verificar si tenía rutina ese día
  const semana = obtenerLunesDeFecha(fecha);
  const diaSemana = obtenerDiaSemana(fecha); // lunes, martes, etc.
  
  const rutina = await fetch(`/api/rutina/${clienteId}/${semana}`, {
    headers: { Authorization: `Bearer ${token}` }
  }).then(r => r.json()).catch(() => null);
  
  const tienePlan = rutina && rutina.dias && 
    rutina.dias.some(d => d.dia_semana === diaSemana && d.ejercicios?.length > 0);
  
  if (!tienePlan) {
    return { estado: 'descanso', color: 'gris', texto: 'Descanso' };
  }
  
  // Si tenía plan, verificar si lo completó
  const checkin = await fetch(`/api/checkin/${clienteId}/${fecha}`, {
    headers: { Authorization: `Bearer ${token}` }
  }).then(r => r.json()).catch(() => null);
  
  const porcentaje = checkin?.porcentaje_adherencia || 0;
  
  if (porcentaje >= 80) return { estado: 'completado', color: 'verde', texto: 'Completado' };
  if (porcentaje >= 40) return { estado: 'parcial', color: 'amarillo', texto: 'Parcial' };
  
  const fechaObj = new Date(fecha);
  const hoy = new Date();
  hoy.setHours(0,0,0,0);
  
  if (fechaObj < hoy) return { estado: 'fallado', color: 'rojo', texto: 'Fallado' };
  return { estado: 'pendiente', color: 'gris', texto: 'Pendiente' };
}

Aplica esta lógica en AMBAS versiones del calendario (semanal y mensual).

PROBLEMA 2 — PANTALLA NEGRA AL NAVEGAR

En TODOS los dashboards (cliente, coach, admin) aplicar este patrón:

// Función global de navegación — NUNCA usar window.location.href entre secciones
function navegarA(seccion) {
  // 1. Ocultar todas las secciones
  document.querySelectorAll('[data-seccion]').forEach(el => {
    el.style.display = 'none';
  });
  
  // 2. Mostrar la sección seleccionada
  const target = document.querySelector(`[data-seccion="${seccion}"]`);
  if (target) {
    target.style.display = 'block';
    target.style.opacity = '0';
    setTimeout(() => { target.style.opacity = '1'; target.style.transition = 'opacity 0.2s'; }, 10);
  }
  
  // 3. Actualizar menú activo
  document.querySelectorAll('.menu-item').forEach(item => {
    item.classList.toggle('activo', item.dataset.ir === seccion);
  });
  
  // 4. Cargar datos frescos
  cargarDatos(seccion);
}

// Cada sección carga sus datos independientemente
async function cargarDatos(seccion) {
  const token = localStorage.getItem('nahofitToken');
  const usuario = JSON.parse(localStorage.getItem('nahofitUsuario') || '{}');
  if (!token || !usuario.id) return;
  
  const contenedor = document.querySelector(`[data-seccion="${seccion}"]`);
  if (!contenedor) return;
  
  // Mostrar spinner
  contenedor.innerHTML = `
    <div style="display:flex;align-items:center;justify-content:center;padding:60px;color:#4B7A5E;">
      <div>Cargando...</div>
    </div>`;
  
  try {
    switch(seccion) {
      case 'dashboard': await cargarDashboard(usuario.id, token, contenedor); break;
      case 'rutina': await cargarRutina(usuario.id, token, contenedor); break;
      case 'nutricion': await cargarNutricion(usuario.id, token, contenedor); break;
      case 'calendario': await cargarCalendario(usuario.id, token, contenedor); break;
      case 'progreso': await cargarProgreso(usuario.id, token, contenedor); break;
      case 'emocional': await cargarEmocional(usuario.id, token, contenedor); break;
      case 'mensajes': await cargarMensajes(usuario.id, token, contenedor); break;
      case 'plan': await cargarPlan(usuario.id, token, contenedor); break;
      case 'pagos': await cargarPagos(usuario.id, token, contenedor); break;
      // COACH
      case 'clientes': await cargarClientes(token, contenedor); break;
      case 'rutinas': await cargarGestionRutinas(token, contenedor); break;
      case 'nutricion-coach': await cargarGestionNutricion(token, contenedor); break;
      case 'reportes': await cargarReportes(token, contenedor); break;
    }
  } catch(e) {
    contenedor.innerHTML = `
      <div style="text-align:center;padding:60px;color:#4B7A5E;">
        <div style="font-size:32px;margin-bottom:16px;">⚠️</div>
        <div style="color:#ECFDF5;margin-bottom:8px;">Error al cargar esta sección</div>
        <button onclick="cargarDatos('${seccion}')" 
          style="background:#10B981;color:#000;border:none;padding:8px 20px;border-radius:6px;cursor:pointer;margin-top:8px;">
          Reintentar
        </button>
      </div>`;
  }
}
