async function abrirConversacion(elemento) {

  const clienteNombre = elemento.dataset.clienteNombre;
  
  // Limpiar panel derecho ANTES de cargar

  // Marcar como activo
  document.querySelectorAll('.conversacion-item').forEach(i => i.classList.remove('active'));
  elemento.classList.add('active');

    document.getElementById('panel-mensajes').innerHTML = '<div style="text-align:center;padding:40px;color:#EF4444;">Error al cargar mensajes</div>';

PROBLEMA 2 — PANTALLA NEGRA AL NAVEGAR (PROBLEMA RAÍZ)
═══════════════════════════════════════

Este es el problema más crítico de toda la app. Al hacer clic en cualquier sección del menú la pantalla queda negra y hay que presionar F5.

CAUSA RAÍZ: La app está usando navegación por URL (window.location.href) en lugar de mostrar/ocultar secciones. Cada vez que cambia la URL, el navegador intenta cargar una nueva página pero el servidor devuelve el mismo HTML y el JavaScript no se reinicializa correctamente.

SOLUCIÓN DEFINITIVA — Single Page Application (SPA) pura:

REGLA: NUNCA usar window.location.href para navegar entre secciones. SIEMPRE usar show/hide de divs.

1. Estructura HTML correcta — UN solo archivo con todas las secciones:
<body>
  <div id="app">
    <nav id="sidebar">...</nav>
    <main id="contenido-principal">
      <div id="seccion-dashboard" class="seccion"></div>
      <div id="seccion-rutina" class="seccion" style="display:none"></div>
      <div id="seccion-nutricion" class="seccion" style="display:none"></div>
      <div id="seccion-progreso" class="seccion" style="display:none"></div>
      <div id="seccion-emocional" class="seccion" style="display:none"></div>
      <div id="seccion-mensajes" class="seccion" style="display:none"></div>
      <div id="seccion-plan" class="seccion" style="display:none"></div>
      <div id="seccion-pagos" class="seccion" style="display:none"></div>
    </main>
  </div>
</body>

2. Función de navegación:
function navegarA(nombreSeccion) {
  // Ocultar todas las secciones
  document.querySelectorAll('.seccion').forEach(s => {
    s.style.display = 'none';
  });
  
  // Mostrar la sección seleccionada
  const seccion = document.getElementById(`seccion-${nombreSeccion}`);
  if (seccion) {
    seccion.style.display = 'block';
  }
  
  // Actualizar menú activo
  document.querySelectorAll('.menu-item').forEach(m => m.classList.remove('active'));
  const menuItem = document.querySelector(`[data-seccion="${nombreSeccion}"]`);
  if (menuItem) menuItem.classList.add('active');
  
  // Cargar datos de esa sección
  cargarDatosSeccion(nombreSeccion);
}

3. Cada menú item usa data-seccion y onclick:
<div class="menu-item" data-seccion="rutina" onclick="navegarA('rutina')">
  Mi Rutina
</div>

async function cargarDatosSeccion(nombre) {
  const token = localStorage.getItem('nahofitToken');
  const usuario = JSON.parse(localStorage.getItem('nahofitUsuario') || '{}');
  
  switch(nombre) {
    case 'dashboard':
      await cargarDashboard(usuario.id, token);
      break;
    case 'rutina':
      await cargarRutinaSemana(usuario.id, token);
      break;
    case 'nutricion':
      await cargarNutricionSemana(usuario.id, token);
      break;
    case 'progreso':
      await cargarProgreso(usuario.id, token);
      break;
    case 'emocional':
      await cargarEmocional(usuario.id, token);
      break;
    case 'mensajes':
      await cargarListaMensajes(usuario.id, token);
      break;
    case 'plan':
      await cargarMiPlan(usuario.id, token);
      break;
    case 'pagos':
      await cargarMisPagos(usuario.id, token);
      break;
  }
}
