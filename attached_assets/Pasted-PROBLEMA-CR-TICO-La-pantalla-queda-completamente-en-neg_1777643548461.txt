<body>
  <div class="login-box">
    <div class="logo">NAHOFIT<span>.LIFE</span></div>

    <div class="error" id="error-msg"></div>
    <div class="field">
      <label>Correo electrónico</label>

  <script>
    // Verificar si ya hay sesión activa
    const tokenExistente = localStorage.getItem('nahofitToken');
    const rolExistente = localStorage.getItem('nahofitRol');
    if (tokenExistente && rolExistente) {
      if (rolExistente === 'administrador') window.location.href = '/admin';
      else if (rolExistente === 'coach') window.location.href = '/coach';
      else window.location.href = '/cliente';
    }

    // Login con Enter
    document.getElementById('password').addEventListener('keypress', function(e) {
      if (e.key === 'Enter') iniciarSesion();
    });

    async function iniciarSesion() {
      const email = document.getElementById('email').value.trim();
      const password = document.getElementById('password').value;
      const errorMsg = document.getElementById('error-msg');
      const btn = document.getElementById('btn-login');

      errorMsg.style.display = 'none';

      if (!email || !password) {
        errorMsg.textContent = 'Por favor ingresa tu email y contraseña.';
        errorMsg.style.display = 'block';
        return;
      }

      btn.textContent = 'Iniciando sesión...';
      btn.classList.add('loading');

      try {
        const res = await fetch('/api/auth/login', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ email, password })
        });

        const data = await res.json();

        if (!res.ok || data.error) {
          errorMsg.textContent = data.error || 'Email o contraseña incorrectos.';
          errorMsg.style.display = 'block';
          btn.textContent = 'Iniciar Sesión';
          btn.classList.remove('loading');
          return;
        }

        // Guardar sesión
        localStorage.setItem('nahofitToken', data.token);
        localStorage.setItem('nahofitRol', data.usuario.rol);
        localStorage.setItem('nahofitUsuario', JSON.stringify(data.usuario));

        // Redirigir según rol
        if (data.usuario.rol === 'administrador') window.location.href = '/admin';
        else if (data.usuario.rol === 'coach') window.location.href = '/coach';
        else window.location.href = '/cliente';

      } catch(e) {
        errorMsg.textContent = 'Error de conexión. Verifica tu internet e intenta de nuevo.';
        errorMsg.style.display = 'block';
        btn.textContent = 'Iniciar Sesión';
        btn.classList.remove('loading');
      }
    }
  </script>
</body>
