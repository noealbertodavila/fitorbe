Función de navegación definitiva:
function irA(seccion) {
  document.querySelectorAll('.seccion').forEach(s => s.style.display = 'none');
  const target = document.querySelector(`[data-seccion="${seccion}"]`);
  if (target) {
    target.style.display = 'block';
    target.style.animation = 'fadeIn 0.2s ease';
  }
  document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('activo'));
  document.querySelector(`[data-ir="${seccion}"]`)?.classList.add('activo');
  cargarSeccion(seccion);
}

@keyframes fadeIn {
  from { opacity: 0; transform: translateY(4px); }
  to { opacity: 1; transform: translateY(0); }
}

Cada sección maneja su propio estado:
async function cargarSeccion(nombre) {
  const contenedor = document.querySelector(`[data-seccion="${nombre}"]`);
  if (!contenedor) return;
  const token = localStorage.getItem('nahofitToken');
  const usuario = JSON.parse(localStorage.getItem('nahofitUsuario') || '{}');
  if (!token || !usuario.id) { window.location.href = '/'; return; }
  
  contenedor.innerHTML = '<div class="cargando"><div class="spinner"></div><span>Cargando...</span></div>';
  
  try {
    switch(nombre) {
      case 'dashboard': await renderDashboard(contenedor, usuario, token); break;
      case 'rutina': await renderRutina(contenedor, usuario, token); break;
      case 'nutricion': await renderNutricion(contenedor, usuario, token); break;
      case 'calendario': await renderCalendario(contenedor, usuario, token); break;
      case 'progreso': await renderProgreso(contenedor, usuario, token); break;
      case 'emocional': await renderEmocional(contenedor, usuario, token); break;
      case 'mensajes': await renderMensajes(contenedor, usuario, token); break;
      case 'plan': await renderPlan(contenedor, usuario, token); break;
      case 'pagos': await renderPagos(contenedor, usuario, token); break;
    }
  } catch(e) {
    contenedor.innerHTML = `
      <div class="error-seccion">
        <div class="error-icono">⚠️</div>
        <div class="error-titulo">Error al cargar</div>
        <button onclick="irA('${nombre}')" class="btn-reintentar">Reintentar</button>
      </div>`;
  }
}

    async function login() {
      const email = document.getElementById('email').value.trim();
      const pass = document.getElementById('pass').value;
      const err = document.getElementById('error-msg');
      const btn = document.getElementById('btn');
      
      err.style.display = 'none';
      if (!email || !pass) {
        err.textContent = 'Ingresa tu correo y contraseña.';
        err.style.display = 'block';
        return;
      }
      
      btn.textContent = 'Ingresando...';
      btn.disabled = true;
      
      try {
        const res = await fetch('/api/auth/login', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ email, password: pass })
        });
        const data = await res.json();
        
        if (!res.ok || data.error) {
          err.textContent = data.error || 'Correo o contraseña incorrectos.';
          err.style.display = 'block';
          btn.textContent = 'Ingresar';
          btn.disabled = false;
          return;
        }
        
        localStorage.setItem('nahofitToken', data.token);
        localStorage.setItem('nahofitRol', data.usuario.rol);
        localStorage.setItem('nahofitUsuario', JSON.stringify(data.usuario));
        
        if (data.usuario.rol === 'administrador') window.location.href = '/admin';
        else if (data.usuario.rol === 'coach') window.location.href = '/coach';
        else window.location.href = '/cliente';
        
      } catch(e) {
        err.textContent = 'Error de conexión. Intenta de nuevo.';
        err.style.display = 'block';
        btn.textContent = 'Ingresar';
        btn.disabled = false;
      }
    }
  </script>
